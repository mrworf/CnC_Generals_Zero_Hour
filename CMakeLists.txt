cmake_minimum_required(VERSION 3.25)
project(ZeroHourLinuxQualification LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(ZH_BUILD_RENDERER "Build stock renderer qualification (not required for headless core)" ON)
option(ZH_SANITIZE "Instrument repository-owned code with ASan/UBSan" OFF)
if(ZH_BUILD_RENDERER)
set(ZH_UPSTREAM_ROOT "${CMAKE_SOURCE_DIR}/build/upstream" CACHE PATH "Pristine upstream sibling source root")
set(ZH_UPSTREAM_COMPILER gcc CACHE STRING "Upstream SDK compiler")
set(ZH_UPSTREAM_CONFIG Debug CACHE STRING "Upstream SDK configuration: Debug or Release")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/upstream_bgfx.py"
  verify --source-root "${ZH_UPSTREAM_ROOT}" RESULT_VARIABLE verify_result)
if(NOT verify_result EQUAL 0)
  message(FATAL_ERROR "Pristine locked upstream sources required; see docs/build.md")
endif()
set(BGFX_BIN "${ZH_UPSTREAM_ROOT}/bgfx/.build/linux64_${ZH_UPSTREAM_COMPILER}/bin")
set(BGFX_LIBRARY "${BGFX_BIN}/libbgfx-shared-lib${ZH_UPSTREAM_CONFIG}.so")
set(BGFX_SHADERC "${BGFX_BIN}/shaderc${ZH_UPSTREAM_CONFIG}")
if(NOT EXISTS "${BGFX_LIBRARY}" OR NOT EXISTS "${BGFX_SHADERC}")
  message(FATAL_ERROR "Build stock runtime/shaderc first: python3 tools/upstream_bgfx.py build")
endif()
add_library(stock_bgfx SHARED IMPORTED)
set_target_properties(stock_bgfx PROPERTIES IMPORTED_LOCATION "${BGFX_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${ZH_UPSTREAM_ROOT}/bgfx/include;${ZH_UPSTREAM_ROOT}/bx/include")
set(SHADER_OUTPUTS)
foreach(shader IN ITEMS vs_qualify fs_solid fs_texture fs_terrain fs_flat vs_project fs_project fs_array)
  if(shader MATCHES "^vs_")
    set(stage vertex)
  else()
    set(stage fragment)
  endif()
  set(output "${CMAKE_BINARY_DIR}/shaders/${shader}.bin")
  add_custom_command(OUTPUT "${output}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/shaders"
    COMMAND "${BGFX_SHADERC}" -f "${CMAKE_SOURCE_DIR}/shaders/qualification/${shader}.sc"
      -o "${output}" --type "${stage}" --platform linux -p spirv
      --varyingdef "${CMAKE_SOURCE_DIR}/shaders/qualification/varying.def.sc"
      -i "${ZH_UPSTREAM_ROOT}/bgfx/src" --Werror
    DEPENDS "shaders/qualification/${shader}.sc" shaders/qualification/varying.def.sc
      "${BGFX_SHADERC}" VERBATIM)
  list(APPEND SHADER_OUTPUTS "${output}")
endforeach()
add_custom_target(qualification_shaders DEPENDS ${SHADER_OUTPUTS})
add_executable(renderer_qualification tests/renderer/qualification.cpp)
target_compile_features(renderer_qualification PRIVATE cxx_std_20)
target_compile_options(renderer_qualification PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(renderer_qualification PRIVATE stock_bgfx)
if(ZH_SANITIZE)
  target_compile_options(renderer_qualification PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(renderer_qualification PRIVATE -fsanitize=address,undefined)
endif()
add_dependencies(renderer_qualification qualification_shaders)
find_package(SDL3 REQUIRED CONFIG)
add_executable(renderer_lifecycle tests/renderer/lifecycle.cpp)
target_compile_features(renderer_lifecycle PRIVATE cxx_std_20)
target_compile_options(renderer_lifecycle PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(renderer_lifecycle PRIVATE stock_bgfx SDL3::SDL3)
add_dependencies(renderer_lifecycle qualification_shaders)
if(ZH_SANITIZE)
  target_compile_options(renderer_lifecycle PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(renderer_lifecycle PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME upstream_provenance_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/toolchain/test_upstream_bgfx.py")
option(ZH_ISOLATED_BUS "Isolate only qualification child processes from host D-Bus" OFF)
if(ZH_ISOLATED_BUS)
  set(BUS_ARGUMENT --isolated-bus)
endif()
foreach(family IN ITEMS init admission clears stencil mips filters ordering materials projection formats all)
  add_test(NAME renderer_${family} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/run_gpu_qualification.py"
    "$<TARGET_FILE:renderer_qualification>" "${CMAKE_BINARY_DIR}/shaders" "${CMAKE_BINARY_DIR}/${family}.log" "${family}" ${BUS_ARGUMENT})
  set_tests_properties(renderer_${family} PROPERTIES LABELS "gpu;vulkan;qualification" TIMEOUT 120)
endforeach()
foreach(family IN ITEMS capacity draw-boundary uploads lifetimes presentation-init presentation)
  add_test(NAME renderer_${family} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/run_gpu_qualification.py"
    "$<TARGET_FILE:renderer_lifecycle>" "${CMAKE_BINARY_DIR}/shaders" "${CMAKE_BINARY_DIR}/${family}.log" "${family}" ${BUS_ARGUMENT})
  set_tests_properties(renderer_${family} PROPERTIES LABELS "gpu;vulkan;qualification;lifecycle" TIMEOUT 120)
endforeach()
endif()
include(cmake/OriginalCore.cmake)
